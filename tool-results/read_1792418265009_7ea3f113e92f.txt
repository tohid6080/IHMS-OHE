    58→function ProgressSteps({ step }: { step: 1 | 2 }) {
    59→  const labels = ['مشخصات فردی', 'اطلاعات مراجعه']
    60→  return (
    61→    <div className="flex flex-wrap items-center gap-2">
    62→      {labels.map((label, i) => {
    63→        const n = (i + 1) as 1 | 2
    64→        const active = step === n
    65→        const done = step > n
    66→        return (
    67→          <div key={label} className="flex items-center gap-2">
    68→            <div
    69→              className={`flex size-8 items-center justify-center rounded-full border-2 text-sm font-bold transition-colors ${
    70→                done
    71→                  ? 'border-emerald-500 bg-emerald-500 text-white'
    72→                  : active
    73→                    ? 'border-teal-600 bg-teal-600 text-white'
    74→                    : 'border-slate-300 bg-white text-slate-400'
    75→              }`}
    76→            >
    77→              {done ? <CheckCircle2 className="size-4" /> : toFa(n)}
    78→            </div>
    79→            <span className={`text-sm ${active ? 'font-medium text-foreground' : 'text-muted-foreground'}`}>{label}</span>
    80→            {i < labels.length - 1 && <span className="h-px w-6 bg-border sm:w-14" />}
    81→          </div>
    82→        )
    83→      })}
    84→    </div>
    85→  )
    86→}

    90→
    91→  const [step, setStep] = useState<1 | 2>(1)
    92→  const [busy, setBusy] = useState(false)
    93→  const [searching, setSearching] = useState(false)

    96→

    99→  const [firstName, setFirstName] = useState('')
   100→  const [lastName, setLastName] = useState('')
   101→  const [fatherName, setFatherName] = useState('')
   102→  const [gender, setGender] = useState('MALE')
   103→  const [birthYear, setBirthYear] = useState('')
   104→  const [birthDate, setBirthDate] = useState('')
   105→  const [maritalStatus, setMaritalStatus] = useState('')
   106→  const [childrenCount, setChildrenCount] = useState('')
   107→  const [education, setEducation] = useState('')
   108→  const [phone, setPhone] = useState('')
   109→  const [insuranceNo, setInsuranceNo] = useState('')
   110→  const [militaryStatus, setMilitaryStatus] = useState('')
   111→  const [militaryUnit, setMilitaryUnit] = useState('')
   112→  const [militaryExemption, setMilitaryExemption] = useState('')

   149→        // پیش‌پر کردن فرم از پرونده قبلی

   153→        setFatherName(exact.fatherName || '')

   185→        const scale = Math.min(1, max / Math.max(img.width, img.height))
   186→        const w = Math.round(img.width * scale)
   187→        const h = Math.round(img.height * scale)

   189→        canvas.width = w
   190→        canvas.height = h
   191→        const ctx = canvas.getContext('2d')
   192→        if (!ctx) return
   193→        ctx.drawImage(img, 0, 0, w, h)
   194→        setPhoto(canvas.toDataURL('image/jpeg', 0.85))

   254→          fatherName: fatherName.trim() || undefined,

   257→          birthDate: birthDate || undefined,
   258→          maritalStatus: maritalStatus || undefined,
   259→          childrenCount: childrenCount || undefined,
   260→          education: education || undefined,

   293→            <UserPlus className="size-5 text-teal-600" />
   294→            پذیرش جدید

   296→          <p className="text-sm text-muted-foreground">تشکیل پرونده پزشکی شاغل</p>
   297→        </div>
   298→        <ProgressSteps step={step} />
   299→      </div>
   300→
   301→      {step === 1 && (
   302→        <Card>
   303→          <CardHeader>
   304→            <CardTitle className="text-base">مشخصات فردی</CardTitle>
   305→          </CardHeader>
   306→          <CardContent className="space-y-4">
   307→            {foundPerson && (
   308→              <Alert className="border-emerald-200 bg-emerald-50 text-emerald-800">
   309→                <CheckCircle2 className="size-4" />
   310→                <AlertTitle>پرونده قبلی یافت شد</AlertTitle>
   311→                <AlertDescription>
   312→                  {toFa(foundPerson.casesCount)} مراجعه ثبت‌شده — سوابق قابل مشاهده در صفحه پرونده‌ها است.
   313→                </AlertDescription>
   314→              </Alert>
   315→            )}
   316→            {!foundPerson && notFound && (
   317→              <Alert className="border-slate-200 bg-slate-50 text-slate-600">
   318→                <AlertDescription>پرونده قبلی با این کد ملی یافت نشد — اطلاعات جدید ثبت می‌شود.</AlertDescription>
   319→              </Alert>
   320→            )}
   321→
   322→            {/* کد ملی + جست‌وجو */}
   323→            <div className="grid grid-cols-1 gap-3 sm:grid-cols-[1fr_auto] sm:items-end">
   324→              <div className="space-y-1.5">
   325→                <Label>کد ملی (اجباری)</Label>
   326→                <Input
   327→                  value={nationalId}
   328→                  onChange={(e) => setNationalId(toEn(e.target.value).replace(/\D/g, '').slice(0, 12))}
   329→                  inputMode="numeric"
   330→                  dir="ltr"
   331→                  placeholder="مثلاً 1234567890"
   332→                  className="max-w-[240px]"
   333→                />
   334→                {nid && !/^\d{8,12}$/.test(nid) && (
   335→                  <p className="text-xs text-rose-600">کد ملی باید بین ۸ تا ۱۲ رقم باشد</p>
   336→                )}
   337→              </div>
   338→              <Button variant="outline" onClick={searchPerson} disabled={searching}>
   339→                {searching ? <Loader2 className="size-4 animate-spin" /> : <Search className="size-4" />}
   340→                جست‌وجوی پرونده قبلی
   341→              </Button>
   342→            </div>
   343→
   344→            <Separator />
   345→
   346→            {/* عکس پرسنلی + نام */}
   347→            <div className="grid grid-cols-1 gap-4 lg:grid-cols-[auto_1fr]">

   349→                <Label className="text-xs">عکس پرسنلی</Label>
   350→                <div className="flex items-center gap-3">
   351→                  <Avatar className="size-20 border">
   352→                    {photo ? <AvatarImage src={photo} alt="عکس پرسنلی" /> : null}
   353→                    <AvatarFallback className="bg-teal-50 text-teal-700">
   354→                      {firstName ? firstName[0] : <Camera className="size-6" />}
   355→                    </AvatarFallback>
   356→                  </Avatar>
   357→                  <div className="space-y-1">
   358→                    <Input type="file" accept="image/*" onChange={(e) => onPhotoChange(e.target.files?.[0])} className="w-56 text-xs" disabled={busy} />
   359→                    {photo && (
   360→                      <Button type="button" variant="ghost" size="sm" className="h-6 text-xs text-rose-600" onClick={() => setPhoto('')}>
   361→                        <Trash2 className="size-3" /> حذف عکس
   362→                      </Button>
   363→                    )}
   364→                  </div>
   365→                </div>
   366→              </div>
   367→              <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3">
   368→                <div className="space-y-1.5">

   375→                </div>
   376→                <div className="space-y-1.5">
   377→                  <Label>نام پدر</Label>
   378→                  <Input value={fatherName} onChange={(e) => setFatherName(e.target.value)} />
   379→                </div>
   380→                <div className="space-y-1.5">
   381→                  <Label>جنسیت</Label>

   383→                    <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   384→                    <SelectContent dir="rtl">
   385→                      {Object.entries(GENDER_LABELS).map(([k, v]) => (
   386→                        <SelectItem key={k} value={k}>{v}</SelectItem>
   387→                      ))}
   388→                    </SelectContent>
   389→                  </Select>
   390→                </div>
   391→                <div className="space-y-1.5">
   392→                  <Label>سال تولد (۴ رقم)</Label>

   394→                    value={birthYear}
   395→                    onChange={(e) => setBirthYear(toEn(e.target.value).replace(/\D/g, '').slice(0, 4))}
   396→                    inputMode="numeric"
   397→                    dir="ltr"
   398→                    placeholder="1362"
   399→                  />
   400→                </div>
   401→                <div className="space-y-1.5">
   402→                  <Label className="text-xs">تاریخ تولد (اختیاری)</Label>

   421→                    inputMode="numeric"
   422→                    dir="ltr"
   423→                  />
   424→                </div>

   428→                    <SelectTrigger className="w-full"><SelectValue placeholder="انتخاب" /></SelectTrigger>
   429→                    <SelectContent dir="rtl">
   430→                      {EDUCATION_OPTIONS.map((o) => (
   431→                        <SelectItem key={o} value={o}>{o}</SelectItem>

   433→                    </SelectContent>

   436→                <div className="space-y-1.5">
   437→                  <Label>تلفن همراه</Label>
   438→                  <Input
   439→                    value={phone}
   440→                    onChange={(e) => setPhone(toEn(e.target.value).replace(/[^\d+]/g, '').slice(0, 13))}
   441→                    inputMode="tel"
   442→                    dir="ltr"
   443→                  />
   444→                </div>
   445→                <div className="space-y-1.5">
   446→                  <Label>شماره بیمه</Label>
   447→                  <Input
   448→                    value={insuranceNo}
   449→                    onChange={(e) => setInsuranceNo(toEn(e.target.value).replace(/\D/g, '').slice(0, 20))}
   450→                    inputMode="numeric"
   451→                    dir="ltr"
   452→                  />
   453→                </div>
   454→              </div>
   455→            </div>
   456→
   457→            {/* نظام وظیفه — فقط آقایان */}
   458→            {gender === 'MALE' && (
   459→              <div className="grid grid-cols-1 gap-3 rounded-lg border bg-slate-50/40 p-3 sm:grid-cols-3">
   460→                <div className="space-y-1.5">

   463→                    <SelectTrigger className="w-full"><SelectValue placeholder="انتخاب" /></SelectTrigger>

   470→                </div>
