   167→  const [center, setCenter] = useState<{ orgLine1?: string; orgLine2?: string; orgLine3?: string }>(() => {
   168→    try {
   169→      const raw = typeof window !== 'undefined' ? window.localStorage.getItem('hoyateb_center') : null
   170→      return raw ? (JSON.parse(raw) as { orgLine1?: string; orgLine2?: string; orgLine3?: string }) : {}
   171→    } catch {
   172→      return {}
   173→    }
   174→  })
   175→
   176→  useEffect(() => {
   177→    const handler = (e: StorageEvent) => {
   178→      if (e.key === 'hoyateb_center') {
   179→        try {
   180→          const raw = window.localStorage.getItem('hoyateb_center')
   181→          setCenter(raw ? JSON.parse(raw) : {})
   182→        } catch {
   183→          /* تنظیمات نامعتبر */
   184→        }
   185→      }
   186→    }
   187→    window.addEventListener('storage', handler)

   192→  useEffect(() => {

   194→    let alive = true
   195→    apiGet<CaseDetailDTO>(`/api/cases/${printCaseId}`)
   196→      .then((d) => {
   197→        if (alive) setDetail(d)
   198→      })
