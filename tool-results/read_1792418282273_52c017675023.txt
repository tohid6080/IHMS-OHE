    55→    55→function resizePhoto(file: File, done: (dataUrl: string) => void) {
    56→    56→  const reader = new FileReader()
    57→    57→  reader.onload = () => {
    58→    58→    const img = new Image()
    59→    59→    img.onload = () => {
    60→    60→      const max = 240
    61→    61→      const scale = Math.min(1, max / Math.max(img.width, img.height))
    62→    62→      const w = Math.round(img.width * scale)
    63→    63→      const h = Math.round(img.height * scale)
    64→    64→      const canvas = document.createElement('canvas')
    65→    65→      canvas.width = w
    66→    66→      canvas.height = h
    67→    67→      const ctx = canvas.getContext('2d')
    68→    68→      if (!ctx) return
    69→    69→      ctx.drawImage(img, 0, 0, w, h)
    70→    70→      done(canvas.toDataURL('image/jpeg', 0.85))
    71→    71→    }
    72→    72→    img.src = String(reader.result)
    73→    73→  }
    74→    74→  reader.readAsDataURL(file)
    75→    75→}
